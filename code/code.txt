library(xgboost)
library(survival)
library(dplyr)

set.seed(123)

target_total_size <- 200
source_size <- 500
n_sources <- 2
num_replications <- 5
train_ratio <- 0.6
nround <- 500

myobject <- function(preds, dtrain) {
  T <- getinfo(dtrain, "label")
  d <- attr(dtrain, "censor")
  u <- log(T) - preds
  fai <- dnorm(u)
  Fai <- pnorm(u)
  g <- outer(fai, 1 - Fai, "/")
  g <- as.matrix(g)
  grad <- as.numeric(-(d * u) - g %*% (1 - d))
  hess <- as.numeric(d + ((1 - d) %*% g))
  list(grad = grad, hess = hess)
}

evalerror <- function(preds, dtrain) {
  T <- getinfo(dtrain, "label")
  d <- attr(dtrain, "censor")
  u <- log(T) - preds
  fai <- dnorm(u)
  Fai <- pnorm(u)
  n <- d * log((1 / T) * fai)
  m <- (1 - d) * log(1 - Fai)
  list(metric = "deviance", value = -as.numeric(sum(m + n)))
}

generate_data <- function(n, shift_vec) {
  x <- matrix(rnorm(n * 6), ncol = 6)
  v1 <- rnorm(n)
  real_time <- exp(
    sin(x[, 1] + 1 + shift_vec[1]) +
      (1 + shift_vec[2]) * x[, 2] * x[, 3] -
      (1 + shift_vec[3]) * x[, 4]^2 -
      (0.3 + shift_vec[5]) * abs(x[, 5]) -
      (0.2 + shift_vec[6]) * x[, 6] +
      shift_vec[4] * x[, 4] +
      v1
  )
  cens_time <- runif(n, 0, 4)
  status <- as.integer(real_time <= cens_time)
  obs_time <- ifelse(status == 1, real_time, cens_time)
  list(data = x, label = obs_time, censor = status)
}

param <- list(max_depth = 3, eta = 0.035, alpha = 1)

run_compare_experiment <- function(target_total_size, source_size, param, nround, ratio) {
  target_train_size <- round(target_total_size * ratio)
  idx_train <- 1:target_train_size
  idx_test <- (target_train_size + 1):target_total_size
  
  D_target <- generate_data(target_total_size, rep(0, 6))
  D_target_train_xgb <- xgb.DMatrix(D_target$data[idx_train, ], label = D_target$label[idx_train])
  attr(D_target_train_xgb, "censor") <- D_target$censor[idx_train]
  D_target_test_xgb <- xgb.DMatrix(D_target$data[idx_test, ], label = D_target$label[idx_test])
  attr(D_target_test_xgb, "censor") <- D_target$censor[idx_test]
  test_surv <- Surv(D_target$label[idx_test], D_target$censor[idx_test])
  
  shift_s1 <- c(-1, -0.8, -0.5, 0.5, -0.3, 0)
  shift_s2 <- c(1, 0.8, 0.6, -0.6, -0.4, -0.2)
  
  D_s1 <- generate_data(source_size, shift_s1)
  D_s1_xgb <- xgb.DMatrix(D_s1$data, label = D_s1$label)
  attr(D_s1_xgb, "censor") <- D_s1$censor
  
  D_s2 <- generate_data(source_size, shift_s2)
  D_s2_xgb <- xgb.DMatrix(D_s2$data, label = D_s2$label)
  attr(D_s2_xgb, "censor") <- D_s2$censor
  
  md_target <- xgb.train(param, D_target_train_xgb, nrounds = nround, obj = myobject, feval = evalerror, verbose = 0)
  md_s1 <- xgb.train(param, D_s1_xgb, nrounds = nround, obj = myobject, feval = evalerror, verbose = 0)
  md_s2 <- xgb.train(param, D_s2_xgb, nrounds = nround, obj = myobject, feval = evalerror, verbose = 0)
  
  models <- list(md_target, md_s1, md_s2)
  weights <- sapply(models, function(m) {
    p <- predict(m, D_target_train_xgb)
    concordance(Surv(D_target$label[idx_train], D_target$censor[idx_train]) ~ exp(p))$concordance[1]
  })
  w_norm <- weights / sum(weights)
  
  c_target <- concordance(test_surv ~ exp(predict(md_target, D_target_test_xgb)))$concordance[1]
  c_s1 <- concordance(test_surv ~ exp(predict(md_s1, D_target_test_xgb)))$concordance[1]
  c_s2 <- concordance(test_surv ~ exp(predict(md_s2, D_target_test_xgb)))$concordance[1]
  
  p_weighted <- w_norm[1] * predict(md_target, D_target_test_xgb) +
    w_norm[2] * predict(md_s1, D_target_test_xgb) +
    w_norm[3] * predict(md_s2, D_target_test_xgb)
  
  c_weighted <- concordance(test_surv ~ exp(p_weighted))$concordance[1]
  
  data.frame(
    Method = c("Target_Only", "Source1_Only", "Source2_Only", "Weighted"),
    CIndex = c(c_target, c_s1, c_s2, c_weighted)
  )
}

results_raw <- do.call(rbind, lapply(1:num_replications, function(r) {
  res <- run_compare_experiment(target_total_size, source_size, param, nround, train_ratio)
  res$Replication <- r
  res$Target_Size <- target_total_size
  res$Source_Size <- source_size
  res
}))

summary_data <- results_raw %>%
  group_by(Target_Size, Source_Size, Method) %>%
  summarise(Mean_C = mean(CIndex), SD_C = sd(CIndex), .groups = "drop")

write.csv(results_raw, "results_raw.csv", row.names = FALSE)
write.csv(summary_data, "summary_data.csv", row.names = FALSE)

print(summary_data)
